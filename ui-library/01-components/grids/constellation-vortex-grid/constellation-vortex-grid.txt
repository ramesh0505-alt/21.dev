interface ConstellationItem {
    item: { src: string; alt: string };
    startPos: { x: number; y: number; rotate: number };
    endPos: { x: number; y: number; rotate: number };
    delay: number;
}

// Generate deterministic layout positions for fixed-size stable cards
const generateConstellation = (): ConstellationItem[] => {
    const total = GALLERY_ITEMS.length;
    return GALLERY_ITEMS.map((item, i) => {
        // Compute unique off-screen starting scatter coordinates (in vh/vw)
        const angle = (i / total) * Math.PI * 2;
        const scatterRadiusX = 45 + (i % 3) * 12;
        const scatterRadiusY = 40 + (i % 2) * 15;

        const startX = Math.cos(angle) * scatterRadiusX;
        const startY = Math.sin(angle) * scatterRadiusY;

        // Compute final fixed position in a balanced floating grid/cluster around center
        // Responsive grid offsets using vw/vh units
        const gridCols = 4;
        const col = i % gridCols;
        const row = Math.floor(i / gridCols);

        const finalX = (col - 1.5) * 21; // Stable offset in vw
        const finalY = (row - 0.5) * 28; // Stable offset in vh

        return {
            item,
            startPos: { x: startX, y: startY, rotate: (i % 2 === 0 ? 1 : -1) * (25 + i * 5) },
            endPos: { x: finalX, y: finalY, rotate: (i % 3 === 0 ? 2 : -2) * (i % 4) },
            delay: i * 0.04,
        };
    });
};

const CONSTELLATION_ITEMS = generateConstellation();

function ConstellationCard({ data, progress }: { data: ConstellationItem; progress: any }) {
    const { item, startPos, endPos } = data;

    // Map scroll progress to smooth coordinate translations without changing scale
    const x = useTransform(progress, [0, 0.85], [startPos.x, endPos.x]);
    const y = useTransform(progress, [0, 0.85], [startPos.y, endPos.y]);
    const rotate = useTransform(progress, [0, 0.85], [startPos.rotate, endPos.rotate]);

    // Fade in quickly, stay fully opaque, and fade out smoothly at the very end
    const opacity = useTransform(progress, [0, 0.15, 0.8, 1], [0, 1, 1, 0]);

    return (
        <motion.div
            className="absolute left-1/2 top-1/2 will-change-transform w-32 h-44 sm:w-40 sm:h-56 md:w-52 md:h-72 pointer-events-auto"
            style={{
                x: useTransform(x, (val) => `calc(-50% + ${val}vw)`),
                y: useTransform(y, (val) => `calc(-50% + ${val}vh)`),
                rotate,
                opacity,
            }}
        >
            <div className="group relative h-full w-full overflow-hidden rounded-2xl border border-neutral-200 dark:border-neutral-800 bg-white dark:bg-neutral-900 shadow-xl dark:shadow-2xl transition-all duration-300 hover:border-neutral-400 dark:hover:border-neutral-600">
                <img
                    src={item.src}
                    alt={item.alt}
                    className="h-full w-full object-cover transition-transform duration-700 group-hover:scale-105"
                    loading="lazy"
                />
                <div className="absolute inset-0 bg-gradient-to-t from-black/30 via-transparent to-transparent opacity-30 group-hover:opacity-10 pointer-events-none" />
            </div>
        </motion.div>
    );
}

export default function ConstellationVortex() {
    const containerRef = useRef<HTMLDivElement>(null);
    const { scrollYProgress } = useScroll({
        target: containerRef,
        offset: ["start start", "end end"],
    });

    // Silky smooth spring physics for scroll response
    const progress = useSpring(scrollYProgress, {
        stiffness: 55,
        damping: 22,
        restDelta: 0.001,
    });

    const headerOpacity = useTransform(progress, [0, 0.25, 0.75, 0.95], [1, 0, 0, 1]);
    const headerY = useTransform(progress, [0, 0.25], [0, -30]);

    return (
        <section
            ref={containerRef}
            className="relative w-full h-[250vh] bg-white dark:bg-black text-neutral-900 dark:text-white overflow-clip transition-colors duration-300"
        >
            <div className="sticky top-0 h-[100svh] w-full overflow-hidden flex items-center justify-center bg-white dark:bg-black">

                {/* Main Floating Cards Constellation */}
                <div className="absolute inset-0 z-10 pointer-events-none">
                    {CONSTELLATION_ITEMS.map((data, i) => (
                        <ConstellationCard key={i} data={data} progress={progress} />
                    ))}
                </div>

                {/* Center Overlay Header / Typography */}
                <motion.div
                    className="absolute z-20 text-center pointer-events-none px-4 max-w-2xl"
                    style={{ opacity: headerOpacity, y: headerY }}
                >
                    <span className="text-xs uppercase tracking-[0.3em] text-neutral-500 dark:text-neutral-400 font-semibold mb-3 block">
                        Interactive Gallery
                    </span>
                    <h2 className="text-3xl sm:text-5xl md:text-6xl font-extrabold tracking-tight mb-4">
                        Constellation Convergence
                    </h2>
                    <p className="text-sm sm:text-base text-neutral-600 dark:text-neutral-400 max-w-md mx-auto">
                        Scroll down to watch disparate visual elements converge into a unified spatial arrangement at a fixed, stable size.
                    </p>
                </motion.div>
            </div>
        </section>
    );
}
